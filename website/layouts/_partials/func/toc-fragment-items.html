{{- /* Build
  • TOC nodes from Page.Fragments.Headings with title-case labels. Params: headings (slice), start (default 2), end (default 5) */ -}} {{- $start := .start | default 2 -}} {{- $end := .end | default 5 -}} {{- $items := "" -}} {{- range .headings -}} {{- if and (ge .Level $start) (le .Level $end) -}} {{- $child := "" -}} {{- with .Headings -}} {{- $child = partial "func/toc-fragment-items.html" (dict "headings" . "start" $start "end" $end) -}} {{- end -}} {{- $label := partial "func/title-case" .Title -}} {{- $items = printf `%s
  • %s%s
    • ` $items .Level .ID $label (cond $child (printf "
      %s
    " $child) "") -}} {{- else if .Headings -}} {{- $items = printf "%s%s" $items (partial "func/toc-fragment-items.html" (dict "headings" .Headings "start" $start "end" $end)) -}} {{- end -}} {{- end -}} {{- return $items -}}
